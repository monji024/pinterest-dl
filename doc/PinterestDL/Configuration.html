def initialize
  @user_agent = DEFAULT_USER_AGENT
  @cookies = nil
  @open_timeout = 10
  @read_timeout = 20
  @max_retries = 3
  @retry_wait = 1.0
  @quality = :originals
  @rate_limit_interval = 0.0
  @logger = Logger.new($stdout).tap do |log|
    log.level = Logger::WARN
    log.formatter = proc { |severity, _time, _progname, msg| "[pinterest-dl] #{severity}: #{msg}\n" }
  end
end
